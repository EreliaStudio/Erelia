cmake_minimum_required(VERSION 3.25)
project(Erelia VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(sparkle CONFIG REQUIRED COMPONENTS Core)
add_subdirectory(src/shared)
add_subdirectory(apps/router)
add_subdirectory(apps/server-a)
add_subdirectory(apps/server-b)
if(WIN32)
    find_package(sparkle CONFIG REQUIRED COMPONENTS Graphics)
    add_subdirectory(apps/client)
endif()
if(BUILD_TESTING)
    find_package(GTest CONFIG REQUIRED)
    include(GoogleTest)
    function(erelia_test_suite name family)
        add_executable(${name}Tests tests/${family}.cpp)
        target_link_libraries(${name}Tests PRIVATE GTest::gtest_main ${ARGN})
        gtest_discover_tests(${name}Tests TEST_PREFIX "${family}." PROPERTIES LABELS "${family}")
    endfunction()
    erelia_test_suite(Router Router EreliaRouter)
    erelia_test_suite(ServerA ServerA EreliaServerA)
    erelia_test_suite(ServerB ServerB EreliaServerB)
    if(WIN32)
        erelia_test_suite(Client Client EreliaClientLibrary)
    endif()
endif()
