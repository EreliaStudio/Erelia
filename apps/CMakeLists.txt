# Each new executable owns its CMakeLists.txt. No root CMake edits are needed.
file(GLOB ERELIA_APPLICATION_CMAKELISTS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*/CMakeLists.txt")
foreach(ERELIA_APPLICATION_CMAKELIST IN LISTS ERELIA_APPLICATION_CMAKELISTS)
    get_filename_component(ERELIA_APPLICATION_DIR "${ERELIA_APPLICATION_CMAKELIST}" DIRECTORY)
    get_filename_component(ERELIA_APPLICATION_NAME "${ERELIA_APPLICATION_DIR}" NAME)
    if(NOT ERELIA_BUILD_GRAPHICS AND ERELIA_APPLICATION_NAME STREQUAL "client")
        continue()
    endif()
    add_subdirectory("${ERELIA_APPLICATION_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/${ERELIA_APPLICATION_NAME}")
endforeach()
